// Resultado esperado al ejecutar el binario (exit code de main):
// a = 10, b = 3, c = 2
// base = 5
// return base + (-a + b * c - (a % b)) + 20 = 5 + (-10 + 6 - 1) + 20 = 20

int base;

int main() {
    int a, b, c, res;
    base = 5;
    a = 10;
    b = 3;
    c = 2;

    res = -a + b * c - (a % b);
    return base + res + 20;
}
