// Test ASM 04: Operaciones combinadas con division entera, resto (%) y booleanos.
// Calcula (37 / 5) * 2 + (37 % 5) = 7 * 2 + 2 = 16, y si es_impar(3) suma 3 -> 19.
// Exit code esperado: 19.

boolean es_impar(int n) {
    return !((n % 2) == 0);
}

int calcular(int a, int b) {
    int cociente, resto;
    cociente = a / b;
    resto = a % b;
    return cociente * 2 + resto;
}

int main() {
    int res;
    res = calcular(37, 5);
    if (es_impar(3)) {
        res = res + 3;
    }
    return res;
}
